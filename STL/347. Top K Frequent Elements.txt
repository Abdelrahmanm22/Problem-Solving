class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        map<int, int> mp;
        for(auto i : nums) {
            mp[i]++;
        }
        vector<pair<int, int>> vec(mp.begin(), mp.end());
        
        sort(vec.begin(), vec.end(), [](const pair<int, int> &a, const pair<int, int> &b) {
            return a.second > b.second;
        });
        vector<int> ans;
        for(auto i : vec) {
            if (k>0) {
                ans.push_back(i.first);
            }
            k--;
        }
        return ans;
    }
};
