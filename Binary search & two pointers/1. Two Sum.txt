class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        vector<pair<int,int>>withIndeces;
        vector<int>ans;
        for(int i = 0 ; i < nums.size() ; i++) {
            withIndeces.push_back({nums[i], i});
        }
        sort(withIndeces.begin(), withIndeces.end());
        int l=0,r=nums.size()-1;
        while (l<r) {
            if(withIndeces[l].first+withIndeces[r].first==target) {
                ans.push_back(withIndeces[l].second);
                ans.push_back(withIndeces[r].second);
                break;
            }else if(withIndeces[l].first+withIndeces[r].first>target) {
                r--;
            }else {
                l++;
            }
        }
        return ans;
    }
};
